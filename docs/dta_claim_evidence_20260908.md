import math

# 稀有事件：概率 0.001，1000 次独立抽样。
print((1 - 0.001) ** 1000)
# 0.36769542477096373

# 1000 次独立抽样零失效：二项模型单侧 95% 上界。
print(1 - 0.05 ** (1 / 1000))
# 0.0029912495450953314

# 独立审计样本 10000：单侧有界变量浓缩扣除量。
print(math.sqrt(math.log(1 / 0.05) / (2 * 10000)))
# 0.012238734153404082

# 两点支持域全覆盖，但经验比例从 50:50 变成 99:1。
print(0.5 * (abs(0.99 - 0.5) + abs(0.01 - 0.5)))
# 0.49
